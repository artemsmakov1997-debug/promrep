    function nowHHMM(){
      const d = new Date();
      const hh = String(d.getHours()).padStart(2,'0');
      const mm = String(d.getMinutes()).padStart(2,'0');
      return `${hh}:${mm}`;
    }

    function escapeHtml(s){
      return String(s)
        .replaceAll("&","&amp;")
        .replaceAll("<","&lt;")
        .replaceAll(">","&gt;")
        .replaceAll('"',"&quot;")
        .replaceAll("'","&#039;");
    }

    function matchesFilter(a){
      const tabOk = (state.tab === "all") || (a.tab === state.tab);
      const liveOk = !state.liveOnly || a.live;
      const q = state.query.trim().toLowerCase();
      const qOk = !q || (a.title.toLowerCase().includes(q) || a.lead.toLowerCase().includes(q) || a.rubric.toLowerCase().includes(q));
      return tabOk && liveOk && qOk;
    }

    function relatedFor(article){
      const sameTab = ARTICLES.filter(a => a.id !== article.id && a.tab === article.tab);
      const byTags = ARTICLES.filter(a => a.id !== article.id && a.tab !== article.tab && a.tags.some(t => article.tags.includes(t)));
      const merged = [...sameTab, ...byTags];

      const uniq = [];
      const seen = new Set();
      for(const a of merged){
        if(!seen.has(a.id)){ seen.add(a.id); uniq.push(a); }
      }
      return uniq.slice(0, 4);
    }

    function renderGlobalHint(){
      const q = state.query.trim();
      if(!q){
        globalSearchHint.textContent = "Введите запрос…";
        return;
      }

      // Left list count
      const leftCount = ARTICLES.filter(matchesFilter).length;

      // Feed count (respect liveOnly + query)
      const qLower = q.toLowerCase();
      const feedCount = (state.liveOnly ? feedItems.filter(i => i.live) : feedItems)
        .filter(i => i.title.toLowerCase().includes(qLower) || i.rubric.toLowerCase().includes(qLower)).length;

      globalSearchHint.textContent = `Найдено: ${leftCount} материалов · ${feedCount} в ленте`;
    }
